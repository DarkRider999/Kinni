01 · Business Analysis

Competitor Profile: picsandprompts.com

Business Model: Freemium content subscription platform (NOT a peer-to-peer marketplace). The company creates and curates its own AI photo editing and image generation prompts, then gates premium content behind a subscription paywall.

Revenue Streams

  • Free tier — ad-supported, limited prompt access
  • Premium subscription — full prompt library unlock
  • Likely: one-time prompt packs / bundles
  • Affiliate links to AI tools (Midjourney, DALL·E, Lightroom)

Target Audience

  • Photographers using AI editing (Lightroom, Photoshop)
  • Social media content creators
  • Digital artists exploring AI art
  • Marketing teams needing prompt templates
  • Hobbyists learning Midjourney / Stable Diffusion

02 · Website Audit — Competitor Analysis

Homepage Structure

  • Hero section: Headline + CTA to browse prompts, sample prompt cards displayed
  • Category grid: Visual category tiles (Lightroom, Midjourney, DALL·E, etc.)
  • Featured prompts: Curated selection with thumbnail previews
  • Trending / Popular: Social proof section
  • Premium CTA: "Unlock All Prompts" subscription upsell
  • Footer: Links, newsletter, social

Navigation Architecture

Home
├── Browse (All Prompts)
│   ├── By AI Tool (Midjourney, DALL·E, Lightroom, Photoshop, Stable Diffusion)
│   ├── By Style (Cinematic, Portrait, Landscape, Abstract…)
│   └── Free vs Premium filter
├── Categories
├── New / Trending

UX Gaps Identified

❌ Missing Features

  • No prompt preview / try-before-buy
  • No user ratings or reviews
  • No creator profiles or storefronts
  • No favorites/bookmarking system
  • No prompt collections/folders
  • No semantic search (keyword only)
  • No mobile app
  • No API access

✅ Strengths to Match

  • Clean visual card layout
  • Clear free vs premium distinction
  • Good thumbnail imagery
  • Simple onboarding
  • Category-based discovery
  • Fast page loads (static/CDN)

04 · Database Schema (PostgreSQL)

Core Tables — DDL

-- ═══════════════════════════════════
-- USERS & AUTHENTICATION
-- ═══════════════════════════════════

CREATE TABLE users (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username        VARCHAR(50) UNIQUE NOT NULL,
    email           VARCHAR(255) UNIQUE NOT NULL,
    password_hash   VARCHAR(255),
    full_name       VARCHAR(100),
    avatar_url      TEXT,
    bio             TEXT,
    website_url     TEXT,
    social_links    JSONB DEFAULT '{}',
    role            VARCHAR(20) DEFAULT 'user'
                    CHECK (role IN ('user','creator','admin','moderator')),
    tier            VARCHAR(20) DEFAULT 'free'
                    CHECK (tier IN ('free','pro','creator','enterprise')),
    is_verified     BOOLEAN DEFAULT FALSE,
    is_banned       BOOLEAN DEFAULT FALSE,
    email_verified  BOOLEAN DEFAULT FALSE,
    stripe_customer_id VARCHAR(100),
    creator_payout_method JSONB,
    settings        JSONB DEFAULT '{}',
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW(),
    last_login      TIMESTAMPTZ
);

CREATE TABLE oauth_accounts (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id     UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    provider    VARCHAR(30) NOT NULL,
    provider_id VARCHAR(255) NOT NULL,
    access_token TEXT,
    refresh_token TEXT,
    expires_at  TIMESTAMPTZ,
    UNIQUE (provider, provider_id)
);

CREATE TABLE sessions (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id     UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    token_hash  VARCHAR(255) UNIQUE NOT NULL,
    ip_address  INET,
    user_agent  TEXT,
    expires_at  TIMESTAMPTZ NOT NULL,
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- ═══════════════════════════════════
-- PROMPTS (CORE ENTITY)
-- ═══════════════════════════════════

CREATE TABLE categories (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    slug        VARCHAR(100) UNIQUE NOT NULL,
    name        VARCHAR(100) NOT NULL,
    description TEXT,
    parent_id   UUID REFERENCES categories(id),
    icon_url    TEXT,
    cover_image TEXT,
    sort_order  INT DEFAULT 0,
    is_active   BOOLEAN DEFAULT TRUE,
    prompt_count INT DEFAULT 0,
    seo_title   VARCHAR(160),
    seo_description TEXT,
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE ai_tools (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    slug        VARCHAR(100) UNIQUE NOT NULL,
    name        VARCHAR(100) NOT NULL,
    description TEXT,
    logo_url    TEXT,
    website_url TEXT,
    prompt_format_guide TEXT,
    is_active   BOOLEAN DEFAULT TRUE
);

CREATE TABLE tags (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    slug        VARCHAR(100) UNIQUE NOT NULL,
    name        VARCHAR(100) NOT NULL,
    usage_count INT DEFAULT 0
);

CREATE TABLE prompts (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    slug            VARCHAR(255) UNIQUE NOT NULL,
    creator_id      UUID NOT NULL REFERENCES users(id),
    title           VARCHAR(255) NOT NULL,
    description     TEXT,
    prompt_text     TEXT NOT NULL,
    prompt_negative TEXT,
    preview_image   TEXT,
    example_outputs JSONB DEFAULT '[]',
    category_id     UUID REFERENCES categories(id),
    ai_tool_id      UUID REFERENCES ai_tools(id),
    access_level    VARCHAR(20) DEFAULT 'free'
                    CHECK (access_level IN ('free','pro','premium','paid')),
    price           DECIMAL(10,2) DEFAULT 0.00,
    currency        VARCHAR(3) DEFAULT 'USD',
    is_published    BOOLEAN DEFAULT FALSE,
    is_featured     BOOLEAN DEFAULT FALSE,
    is_trending     BOOLEAN DEFAULT FALSE,
    moderation_status VARCHAR(20) DEFAULT 'pending'
                    CHECK (moderation_status IN ('pending','approved','rejected','flagged')),
    view_count      INT DEFAULT 0,
    download_count  INT DEFAULT 0,
    save_count      INT DEFAULT 0,
    rating_avg      DECIMAL(3,2) DEFAULT 0.00,
    rating_count    INT DEFAULT 0,
    version         INT DEFAULT 1,
    parent_id       UUID REFERENCES prompts(id),
    fork_count      INT DEFAULT 0,
    variables       JSONB DEFAULT '[]',
    metadata        JSONB DEFAULT '{}',
    search_vector   TSVECTOR,
    embedding       VECTOR(1536),
    seo_title       VARCHAR(160),
    seo_description TEXT,
    published_at    TIMESTAMPTZ,
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE prompt_tags (
    prompt_id UUID NOT NULL REFERENCES prompts(id) ON DELETE CASCADE,
    tag_id    UUID NOT NULL REFERENCES tags(id) ON DELETE CASCADE,
    PRIMARY KEY (prompt_id, tag_id)
);

CREATE TABLE prompt_versions (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    prompt_id   UUID NOT NULL REFERENCES prompts(id) ON DELETE CASCADE,
    version     INT NOT NULL,
    prompt_text TEXT NOT NULL,
    change_notes TEXT,
    created_by  UUID REFERENCES users(id),
    created_at  TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE (prompt_id, version)
);

-- COMMERCE

CREATE TABLE subscriptions (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id),
    stripe_sub_id   VARCHAR(100) UNIQUE,
    plan            VARCHAR(30) NOT NULL,
    status          VARCHAR(20) DEFAULT 'active',
    current_period_start TIMESTAMPTZ,
    current_period_end   TIMESTAMPTZ,
    cancel_at_period_end BOOLEAN DEFAULT FALSE,
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE orders (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    buyer_id        UUID NOT NULL REFERENCES users(id),
    prompt_id       UUID NOT NULL REFERENCES prompts(id),
    amount          DECIMAL(10,2) NOT NULL,
    platform_fee    DECIMAL(10,2) NOT NULL,
    creator_earning DECIMAL(10,2) NOT NULL,
    currency        VARCHAR(3) DEFAULT 'USD',
    payment_method  VARCHAR(30),
    stripe_payment_intent VARCHAR(100),
    status          VARCHAR(20) DEFAULT 'pending',
    created_at      TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE payouts (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    creator_id      UUID NOT NULL REFERENCES users(id),
    amount          DECIMAL(10,2) NOT NULL,
    currency        VARCHAR(3) DEFAULT 'USD',
    status          VARCHAR(20) DEFAULT 'pending',
    payout_method   VARCHAR(30),
    external_ref    VARCHAR(100),
    period_start    TIMESTAMPTZ,
    period_end      TIMESTAMPTZ,
    created_at      TIMESTAMPTZ DEFAULT NOW()
);

-- ═══════════════════════════════════
-- SOCIAL & COLLECTIONS
-- ═══════════════════════════════════

CREATE TABLE reviews (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    prompt_id   UUID NOT NULL REFERENCES prompts(id) ON DELETE CASCADE,
    user_id     UUID NOT NULL REFERENCES users(id),
    rating      SMALLINT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    title       VARCHAR(100),
    body        TEXT,
    is_verified_purchase BOOLEAN DEFAULT FALSE,
    helpful_count INT DEFAULT 0,
    created_at  TIMESTAMPTZ DEFAULT NOW(),
    updated_at  TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE (prompt_id, user_id)
);

CREATE TABLE collections (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id     UUID NOT NULL REFERENCES users(id),
    name        VARCHAR(100) NOT NULL,
    description TEXT,
    cover_image TEXT,
    is_public   BOOLEAN DEFAULT TRUE,
    prompt_count INT DEFAULT 0,
    follower_count INT DEFAULT 0,
    created_at  TIMESTAMPTZ DEFAULT NOW(),
    updated_at  TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE collection_prompts (
    collection_id UUID NOT NULL REFERENCES collections(id) ON DELETE CASCADE,
    prompt_id     UUID NOT NULL REFERENCES prompts(id) ON DELETE CASCADE,
    sort_order    INT DEFAULT 0,
    added_at      TIMESTAMPTZ DEFAULT NOW(),
    PRIMARY KEY (collection_id, prompt_id)
);

CREATE TABLE favorites (
    user_id     UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    prompt_id   UUID NOT NULL REFERENCES prompts(id) ON DELETE CASCADE,
    created_at  TIMESTAMPTZ DEFAULT NOW(),
    PRIMARY KEY (user_id, prompt_id)
);

CREATE TABLE follows (
    follower_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    following_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    created_at  TIMESTAMPTZ DEFAULT NOW(),
    PRIMARY KEY (follower_id, following_id)
);

-- ═══════════════════════════════════
-- ANALYTICS & ACTIVITY
-- ═══════════════════════════════════

CREATE TABLE prompt_views (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    prompt_id   UUID NOT NULL REFERENCES prompts(id) ON DELETE CASCADE,
    user_id     UUID REFERENCES users(id),
    session_id  UUID,
    ip_address  INET,
    referrer    TEXT,
    viewed_at   TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE user_activity (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id     UUID NOT NULL REFERENCES users(id),
    action      VARCHAR(50) NOT NULL,
    entity_type VARCHAR(50),
    entity_id   UUID,
    metadata    JSONB DEFAULT '{}',
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE notifications (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id     UUID NOT NULL REFERENCES users(id),
    type        VARCHAR(50) NOT NULL,
    title       VARCHAR(255),
    body        TEXT,
    link        TEXT,
    is_read     BOOLEAN DEFAULT FALSE,
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE team_workspaces (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name        VARCHAR(100) NOT NULL,
    owner_id    UUID NOT NULL REFERENCES users(id),
    plan        VARCHAR(30) DEFAULT 'team',
    max_members INT DEFAULT 5,
    created_at  TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE workspace_members (
    workspace_id UUID NOT NULL REFERENCES team_workspaces(id) ON DELETE CASCADE,
    user_id      UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    role         VARCHAR(20) DEFAULT 'member'
                 CHECK (role IN ('owner','admin','editor','viewer')),
    joined_at    TIMESTAMPTZ DEFAULT NOW(),
    PRIMARY KEY (workspace_id, user_id)
);

Key Indexes

-- Search and discovery
CREATE INDEX idx_prompts_search ON prompts USING GIN(search_vector);
CREATE INDEX idx_prompts_embedding ON prompts USING ivfflat(embedding vector_cosine_ops);
CREATE INDEX idx_prompts_category ON prompts(category_id, is_published, created_at DESC);
CREATE INDEX idx_prompts_creator ON prompts(creator_id, is_published);
CREATE INDEX idx_prompts_trending ON prompts(is_trending, rating_avg DESC) WHERE is_published = TRUE;
CREATE INDEX idx_prompts_featured ON prompts(is_featured, published_at DESC) WHERE is_published = TRUE;

-- Commerce
CREATE INDEX idx_orders_buyer ON orders(buyer_id, created_at DESC);
CREATE INDEX idx_orders_creator ON orders(prompt_id) INCLUDE (creator_earning, created_at);

-- Social
CREATE INDEX idx_reviews_prompt ON reviews(prompt_id, created_at DESC);
CREATE INDEX idx_favorites_user ON favorites(user_id, created_at DESC);

-- Analytics
CREATE INDEX idx_views_prompt_date ON prompt_views(prompt_id, viewed_at);
CREATE INDEX idx_activity_user ON user_activity(user_id, created_at DESC);
